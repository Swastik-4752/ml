# ml installer for Windows - after this, type  ml  (or  ml 2) in any cmd / PowerShell window.
# Run:   irm https://swastik-4752.github.io/ml/install.txt | iex
& {
    $Base = 'https://swastik-4752.github.io/ml'
    $cmd = "@echo off`r`npowershell -NoProfile -ExecutionPolicy Bypass -Command `"`$env:ML_N='%~1'; irm '$Base/w.txt' | iex`"`r`n"

    # WindowsApps is already on every user's PATH on Windows 10/11, so ml works at once.
    $dir = Join-Path $env:LOCALAPPDATA 'Microsoft\WindowsApps'
    $onPath = ($env:Path -split ';') -contains $dir
    if (-not ((Test-Path $dir) -and $onPath)) {
        $dir = Join-Path $env:USERPROFILE '.ml'
        New-Item -ItemType Directory -Force -Path $dir | Out-Null
        $userPath = [Environment]::GetEnvironmentVariable('Path', 'User')
        if (($userPath -split ';') -notcontains $dir) {
            [Environment]::SetEnvironmentVariable('Path', ($userPath.TrimEnd(';') + ';' + $dir), 'User')
        }
        $env:Path += ";$dir"
        Write-Host '  (Open a new terminal window if ml is not found.)' -ForegroundColor Yellow
    }

    [IO.File]::WriteAllText((Join-Path $dir 'ml.cmd'), $cmd, (New-Object Text.ASCIIEncoding))
    Write-Host ''
    Write-Host '  Installed!  Now type:' -ForegroundColor Green
    Write-Host '     ml        -> menu of practicals'
    Write-Host '     ml 2      -> straight to practical 2'
    Write-Host ''
}
