# ===== Practical 10 - K-Means clustering (iris) =====

# ----- kmeans.py -----
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.datasets import load_iris
from sklearn.preprocessing import MinMaxScaler
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

iris = load_iris()
df1 = pd.DataFrame(iris.data, columns=iris.feature_names)
df1["species"] = iris.target
print(df1.head())
print(df1["species"].value_counts())

# species is the answer, so drop it; scale the 4 features to 0-1
df = pd.DataFrame(MinMaxScaler().fit_transform(df1.drop("species", axis=1)))

# Elbow method: WCSS for k = 1..10
wcss = []
for k in range(1, 11):
    kmeans = KMeans(n_clusters=k, init="k-means++", random_state=42).fit(df)
    wcss.append(kmeans.inertia_)
plt.plot(range(1, 11), wcss)
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.show()

# Silhouette score for k = 2..10 (higher is better)
scores = []
for k in range(2, 11):
    kmeans = KMeans(n_clusters=k, init="k-means++", random_state=42).fit(df)
    scores.append(silhouette_score(df, kmeans.labels_))
print("Silhouette scores:", scores)
plt.plot(range(2, 11), scores)
plt.xlabel("Number of Clusters")
plt.ylabel("Silhouette score")
plt.show()

kmeans = KMeans(n_clusters=2, init="k-means++", random_state=42)
df["Cluster"] = kmeans.fit_predict(df)
print(df["Cluster"].value_counts())

df1["Cluster"] = kmeans.labels_
print(df1[["species", "Cluster"]].tail(10))
df1.to_excel("iris_cluster_assignments.xlsx", index=False)
print("Saved iris_cluster_assignments.xlsx")

centroids = kmeans.cluster_centers_
plt.scatter(df[2], df[3], c=df["Cluster"], cmap="viridis", alpha=0.6, label="Data Points")
plt.scatter(centroids[:, 2], centroids[:, 3], c="red", marker="X", s=200, edgecolors="black", label="Centroids")
plt.xlabel("Petal Length (scaled)")
plt.ylabel("Petal Width (scaled)")
plt.title("K-Means Clustering on Iris Dataset with Centroids")
plt.legend()
plt.show()
